\def\angleSommet{65}; \def\angleBase{15};\def\demiSommet{1.5}; \def\demiBase{0.2};

% Coordonnées des ailes
\pgfmathsetmacro{\hauteur}{\demiBase * 2.2};  \def\rotation{10};
\pgfmathsetmacro{\aile11}{(\demiBase + \hauteur * sin(\angleBase)) * cos(\rotation)};
\pgfmathsetmacro{\aile21}{(\demiBase + \hauteur * sin(\angleBase)) * cos(60-\rotation)};
\pgfmathsetmacro{\aile31}{(\demiBase + \hauteur * sin(\angleBase)) * cos(60+\rotation)};
\pgfmathsetmacro{\aile12}{\demiBase * cos(\rotation)};
\pgfmathsetmacro{\aile22}{\demiBase * cos(60-\rotation)};

\newcommand\fusee{
% Coordonnées des points
\coordinate (ss) at (\demiSommet,0);
\coordinate (dd) at (-\demiSommet,-\demiBase);
\coordinate (gg) at (-\demiSommet,\demiBase);
% Corps de la fusée
\fill[line width=1.5pt, draw=black, fill=white]
  (gg) to[out=\angleBase, in=90+\angleSommet]
  (ss) to[out=-90-\angleSommet, in=-\angleBase]
  (dd) -- cycle;
cycle;
