% Angles et dimensions corps
\def\angleSommet{65} \def\angleBase{15} \def\demiSommet{1.5} \def\demiBase{0.2}

% Ordonnées des ailes en fonction de rotation
\pgfmathsetmacro{\hauteur}{\demiBase * 2.2}  \def\rotation{15}

\pgfmathsetmacro{\aileHautA}{(\demiBase + \hauteur * sin(\angleBase)) * cos(\rotation)}
\pgfmathsetmacro{\aileAvantA}{-(\demiBase + \hauteur * sin(\angleBase)) * cos(60-\rotation)}
\pgfmathsetmacro{\aileArriereA}{-(\demiBase + \hauteur * sin(\angleBase)) * cos(60+\rotation)}

\pgfmathsetmacro{\aileHautB}{\demiBase * cos(\rotation)}
\pgfmathsetmacro{\aileAvantB}{-\demiBase * cos(60-\rotation)}
\pgfmathsetmacro{\aileArriereB}{-\demiBase * cos(60+\rotation)}

\pgfmathsetmacro{\aileHautC}{\hauteur * cos(\rotation)}
\pgfmathsetmacro{\aileAvantC}{-\hauteur * cos(60-\rotation)}
\pgfmathsetmacro{\aileArriereC}{-\hauteur * cos(60+\rotation)}

\pgfmathsetmacro{\aileHautD}{(\hauteur+\demiBase) * cos(\rotation)}
\pgfmathsetmacro{\aileAvantD}{(-\hauteur-\demiBase) * cos(60-\rotation)}
\pgfmathsetmacro{\aileArriereD}{(-\hauteur-\demiBase) * cos(60+\rotation)}

% Abscisses des ailes
\def\aileA{-\demiSommet+\hauteur} \def\aileC{-\demiSommet-\hauteur}
\def\decalage{0.02} % perspective
% Angles des ailes
\def\anglePied{45}
\tikzstyle{pied}=[ultra thin, draw=black, top color=red , bottom color=gray!60!red]%ultra thin, draw=black, fill=red
\tikzstyle{cheville}=[ultra thin, draw=black, top color=blue , bottom color=gray!60!blue]
\newcommand\fusee{
%  PIED 3
\coordinate (a31) at (\aileA-\decalage,\aileArriereA); \coordinate (a32) at (-\demiSommet-\decalage,\aileArriereB);
\coordinate (a33) at (\aileC-\decalage,\aileArriereC); \coordinate (a34) at (-\demiSommet-\decalage,\aileArriereD);
\shade[pied] (\aileC-\decalage,\aileArriereD) circle (\demiBase);
\shade[cheville]
  (\aileC-\decalage,\aileArriereD + \demiBase) to[out=0, in=90+\anglePied] (a34)
  to[out=-90-\anglePied, in=0] (\aileC-\decalage,\aileArriereD - \demiBase) -- cycle;
\fill[ultra thin, draw=black, fill=red!60!gray, rounded corners=1pt] (a31) -- (a32) -- (a33) -- (a34) -- cycle;

%  CORPS DE LA FUSÉE
\coordinate (ss) at (\demiSommet,0);
\coordinate (dd) at (-\demiSommet,-\demiBase);
\coordinate (gg) at (-\demiSommet,\demiBase);
\shade[top color=blue!60! , bottom color=gray!60!blue]
  (gg) to[out=\angleBase, in=90+\angleSommet] (ss) to[out=-90-\angleSommet, in=-\angleBase] (dd) -- cycle;
\fill[fill=black!40!red] (-\demiSommet,0)ellipse(\decalage cm and \demiBase cm);

%  PIED 1
\coordinate (a11) at (\aileA,\aileHautA); \coordinate (a12) at (-\demiSommet,\aileHautB);
\coordinate (a13) at (\aileC,\aileHautC); \coordinate (a14) at (-\demiSommet,\aileHautD);
\fill[ultra thin, draw=black, fill=red!60!gray] (a11) -- (a12) -- (a13) -- (a14) -- cycle;
\shade[pied] (\aileC,\aileHautD) circle (\demiBase);
\shade[cheville] (\aileC,\aileHautD + \demiBase) to[out=0, in=90+\anglePied]
(a14) to[out=-90-\anglePied, in=0] (\aileC,\aileHautD - \demiBase) -- cycle;

%  PIED 2
\coordinate (a21) at (\aileA+\decalage,\aileAvantA); \coordinate (a22) at (-\demiSommet+\decalage,\aileAvantB);
\coordinate (a23) at (\aileC+\decalage,\aileAvantC); \coordinate (a24) at (-\demiSommet+\decalage,\aileAvantD);
\fill[ultra thin, draw=black, fill=red!80!gray] (a21) -- (a22) -- (a23) -- (a24) -- cycle;
\shade[pied] (\aileC+\decalage,\aileAvantD) circle (\demiBase);
\shade[cheville]
  (\aileC+\decalage,\aileAvantD + \demiBase) to[out=0, in=90+\anglePied] (a24)
  to[out=-90-\anglePied, in=0] (\aileC+\decalage,\aileAvantD - \demiBase) -- cycle;
}
